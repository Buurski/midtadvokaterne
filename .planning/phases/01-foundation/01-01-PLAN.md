**layout.tsx** — replace entire file with:
```tsx
import type { Metadata } from "next";
import { Playfair_Display, Inter } from "next/font/google";
import "./globals.css";

const playfair = Playfair_Display({
  subsets: ["latin"],
  variable: "--font-playfair",
  display: "swap",
});

const inter = Inter({
  subsets: ["latin"],
  variable: "--font-inter",
  display: "swap",
});

export const metadata: Metadata = {
  title: "MidtAdvokaterne A/S — Advokater i Ikast",
  description:
    "MidtAdvokaterne er et erfarent advokatfirma i Ikast. Vi hjælper private og erhvervskunder med juridisk rådgivning siden 1964.",
};

export default function RootLayout({
  children,
}: Readonly<{ children: React.ReactNode }>) {
  return (
    <html
      lang="da"
      className={`${playfair.variable} ${inter.variable} h-full antialiased`}
    >
      <body className="min-h-full flex flex-col font-body bg-off-white text-text">
        {children}
      </body>
    </html>
  );
}
```

Key points:
- `lang="da"` (was "en")
- Both fonts use `variable:` (NOT `className:`) so @theme can reference them
- Geist and Geist_Mono imports removed entirely
- metadata updated to Danish content
- body applies `font-body bg-off-white text-text` (Tailwind utilities from @theme tokens)
